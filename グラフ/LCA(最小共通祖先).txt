struct EulerTour{
  vector<ll> fa,la;//頂点vがDFS探索順で最初に現れた時間とその頂点を抜けた時間
  vector<ll> depth;
  vector<pair<ll,ll>> euler_tour;//dfsしたときに現れる頂点の(深さ、頂点番号)
  EulerTour(){}
  EulerTour(ll n,vector<vector<ll>> &g){
    fa.resize(n);la.resize(n);
    depth.resize(n,INF);
    depth[0] = 0;
    dfs_euler(g,0);
  }

  void dfs_euler(vector<vector<ll>> &g,ll v){
    fa[v] = euler_tour.size();
    euler_tour.push_back({depth[v],v});
    bool leaf = true;
    for(auto &p:g[v]){
      if(depth[p] == INF){
        leaf = false;
        depth[p] = depth[v] + 1;
        dfs_euler(g,p);
        euler_tour.push_back({depth[v],v});
      }
    }
    if(leaf){//葉も2回現れるように
      euler_tour.push_back({depth[v],v});
    }
    la[v] = euler_tour.size()-1;
  }
};

// LCA用のセグ木（深さが最小の頂点を返す）
pair<ll,ll> lca_op(pair<ll,ll> a,pair<ll,ll> b){
  return min(a,b);
}
pair<ll,ll> lca_e(){
  return {INF,INF};//1LL<<60
}

struct LCA{
  EulerTour et;
  atcoder::segtree<pair<ll,ll>,lca_op,lca_e> seg_euler;

  LCA(){}
  LCA(ll n,vector<vector<ll>> &g):et(n,g),seg_euler(et.euler_tour){}

  // x,yのLCA({深さ,頂点番号})
  // uvの距離は find[u]+find[v]-2*find[lca.query(u,v).SE]
  pair<ll,ll> query(ll x,ll y){
    return seg_euler.prod(min(et.fa[x],et.fa[y]),max(et.fa[x],et.fa[y])+1);
  }

  /* uvの距離を求めるやつ

  };

};

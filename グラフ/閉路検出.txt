// directed = true  : 有向グラフ
// directed = false : 無向グラフ（単純グラフを想定）
// 返り値 : 閉路を構成する頂点列があればそれを一つ。閉路がなければ空配列
auto find_cycle = [&] ( bool directed)-> vll {
  vll seen(n, 0);
  vll finish(n, 0);
  vll used; // 現在のDFS上の頂点
  vll loop;

  auto dfs = [&](auto self, ll v, ll par) -> ll {
    seen[v] = 1;
    used.push_back(v);
    for(auto &p : g[v]){
      // 無向グラフでは直前に通った辺を逆向きに戻るのを除外
      // !!多重辺がある場合は頂点parではなくedge idを持つ!!
      if(!directed && p == par) continue;
      if(finish[p]) continue;
      
      if(seen[p] && !finish[p]){// 現在のDFS上の頂点への辺 → 閉路
        return p; // 閉路の始点
      }
      if(!seen[p]){
        ll ret = self(self, p, v);
        if(ret != -1) return ret;
      }
    }

    finish[v] = 1;
    used.pop_back();
    return -1;
  };

  rep(i, n){
    if(seen[i]) continue;
    ll ret = dfs(dfs, i, -1);
    if(ret != -1){
      // used の末尾から ret までが閉路
      while(!used.empty()){
        ll v = used.back();
        used.pop_back();

        loop.push_back(v);
        if(v == ret) break;
      }
      reverse(loop.begin(), loop.end());
      return loop;
    }
  }
  return {};
};

//verify https://atcoder.jp/contests/awc0150/submissions/80782425
struct CentroidDecomposition {
  ll n;
  vector<vector<ll>> g;
  vector<ll> used, subtree_size, parent;

  CentroidDecomposition(ll n, const vector<vector<ll>>& g)
      : n(n), g(g), used(n), subtree_size(n), parent(n, -1) {
        build();
      }

  void build() {
    if (n == 0) return;
    fill(used.begin(), used.end(), 0);
    fill(parent.begin(), parent.end(), -1);

    vector<ll> dfspar(n, -1);
    // (処理する連結成分の始点, 重心分解木の親)
    vector<pair<ll, ll>> st{{0, -1}};

    while (!st.empty()) {
      auto [start, cp] = st.back();
      st.pop_back();

      // DFS順に頂点を列挙する
      vector<ll> order{start};
      dfspar[start] = -1;
      for (ll i = 0; i < (ll)order.size(); ++i) {
        ll v = order[i];
        for (ll u : g[v]) {
          if (used[u] || u == dfspar[v]) continue;
          dfspar[u] = v;
          order.push_back(u);
        }
      }

      // 逆順に部分木サイズを計算
      ll m = order.size();
      for (ll i = m - 1; i >= 0; --i) {
        ll v = order[i];
        subtree_size[v] = 1;
        for (ll u : g[v]) {
          if (!used[u] && dfspar[u] == v)
            subtree_size[v] += subtree_size[u];
        }
      }

      // 重心を探す
      ll centroid = -1;
      for (ll v : order) {
        ll mx = m - subtree_size[v]; // 親側
        for (ll u : g[v]) {
          if (!used[u] && dfspar[u] == v)
            mx = max(mx, subtree_size[u]);
        }
        if (2 * mx <= m) {
          centroid = v;
          break;
        }
      }

      // 重心を確定
      used[centroid] = 1;
      parent[centroid] = cp;

      // 重心を取り除いた各連結成分を処理
      for (ll u : g[centroid]) {
        if (!used[u])
          st.push_back({u, centroid});
      }
    }
  }
};

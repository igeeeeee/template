vector dp(n+1,vector(2,vector<mint>(d)));
reps(k,1,10){//1桁目だけ別処理
  if(k==s[0]){
    dp[1][1][k%d]++;
  }else if(k<s[0]){
    dp[1][0][k%d]++;
  }
}
reps(i,1,n){
  rep(j,d){
    rep(k,10){
      //0->0はいつでもできる
      dp[i+1][0][(j+k)%d]+=dp[i][0][j];
      if(k==s[i]){//1->1が発生
        dp[i+1][1][(j+k)%d]+=dp[i][1][j];
      }else if(k<s[i]){//1->0が発生
        dp[i+1][0][(j+k)%d]+=dp[i][1][j];
      }
    }
  }
  reps(k,1,10){//そこから始まる桁
    dp[i+1][0][k%d]++;
  }
}

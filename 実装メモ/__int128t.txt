using i128 = __int128_t;
using u128 = __uint128_t;

ostream& operator<<(ostream& os, u128 x) {
    if (x >= 10) os << x / 10;
    return os << char('0' + x % 10);
}

ostream& operator<<(ostream& os, i128 x) {
    if (x < 0) {
        os << '-';
        return os << (-(u128)x);
    }
    return os << (u128)x;
}

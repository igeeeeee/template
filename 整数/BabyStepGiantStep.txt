// a^x ≡ b (mod m) となる最小の非負整数 x
// m は素数
// 解が存在しない場合は -1
// verify https://atcoder.jp/contests/arc042/submissions/77422268
ll BabyStepGiantStep(ll a, ll b, ll m) {
  assert(m >= 2);

  a %= m;
  b %= m;
  if (a < 0) a += m;
  if (b < 0) b += m;

  // 0^0 = 1 と扱う
  if (a == 0) {
    if (b == 1) return 0;
    if (b == 0) return 1;
    return -1;
  }

  // a != 0 なので、a^x は常に非零
  if (b == 0) return -1;

  ll B = sqrtl((long double)(m - 1)) + 1;

  unordered_map<ll, ll> mp;
  mp.reserve(B * 2);
  mp.max_load_factor(0.7);

  // baby step: a^j
  ll v = 1;
  rep(j, B) {
    // 同じ値が複数回出た場合、最小の j を残す
    if (!mp.contains(v)) mp[v] = j;
    v = (__int128)v * a % m;
  }

  ll aB = Modlpow(a, B, m);
  ll inv_aB = Modlpow(aB, m - 2, m);

  // need = b * a^(-Bi)
  ll need = b;

  rep(i, B) {
    auto it = mp.find(need);
    if (it != mp.end()) {
      return i * B + it->second;
    }
    need = (__int128)need * inv_aB % m;
  }

  return -1;
}

struct Min2 {
  ll d1 = INF, d2 = INF;
  ll i1 = -1, i2 = -1;

  // 異なる index について、小さい値を2つ保持する
  // 同値の場合は先に入っていたものを優先する
  bool insert(ll d, ll i) {
    if (i == i1) {
      return chmin(d1, d);
    }

    if (d2 <= d) {
      return false;
    }

    d2 = d;
    i2 = i;

    if (d2 < d1) {
      swap(d1, d2);
      swap(i1, i2);
    }

    return true;
  }

  // index i 以外で最小の値
  ll get_except(ll i) const {
    return i1 != i ? d1 : d2;
  }
};


struct Max2 {
  ll d1 = -INF, d2 = -INF;
  ll i1 = -1, i2 = -1;

  // 異なる index について、大きい値を2つ保持する
  // 同値の場合は先に入っていたものを優先する
  bool insert(ll d, ll i) {
    if (i == i1) {
      return chmax(d1, d);
    }

    if (d2 >= d) return false;

    d2 = d;
    i2 = i;

    if (d1 < d2) {
      swap(d1, d2);
      swap(i1, i2);
    }

    return true;
  }

  // index i 以外で最大の値
  ll get_except(ll i) const {
    return i1 != i ? d1 : d2;
  }
};

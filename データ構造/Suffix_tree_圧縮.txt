//verify https://atcoder.jp/contests/abc433/submissions/80464222
// 圧縮 Suffix Tree（全 suffix の Trie を圧縮した木、番兵文字なし）。
// 構築：ACL の suffix_array / lcp_array の計算後 O(N)、空間 O(N)。
struct CompressedSuffixTree {
  struct Node {
    int par = -1, depth = 0;        // 親、根からの文字列長
    int l = 0, r = 0;               // 親からの辺に対応する s[l, r)
    int sa_l = 0, sa_r = 0;         // 部分木の SA 区間 [sa_l, sa_r)
    int suffix_pos = -1;           // ここで終わる suffix の開始位置（なければ -1）
    std::vector<int> ch;           // 子頂点（辞書順）
  };

  std::string s;
  std::vector<int> sa, lcp, rank, suffix_node;
  std::vector<Node> t;
  int root = 0;

  CompressedSuffixTree() { t.emplace_back(); }
  explicit CompressedSuffixTree(std::string s) { build(std::move(s)); }

  void build(std::string str) {
    s = std::move(str);
    int n = (int)s.size();
    sa = atcoder::suffix_array(s);
    lcp = n ? atcoder::lcp_array(s, sa) : std::vector<int>{};
    rank.resize(n);
    for (int i = 0; i < n; ++i) rank[sa[i]] = i;
    suffix_node.assign(n, -1);

    t.clear();
    t.reserve(2 * n + 1);
    t.emplace_back(); // 根は空文字列
    std::vector<int> path = {root};

    for (int i = 0; i < n; ++i) {
      int common = i ? lcp[i - 1] : 0;
      while (t[path.back()].depth > common) path.pop_back();
      int p = path.back();

      if (t[p].depth < common) {
        // 直前の suffix と共有する辺を途中で分割する。
        int old = t[p].ch.back();
        int cut = common - t[p].depth;
        int mid = (int)t.size();
        t.emplace_back();
        t[mid].par = p;
        t[mid].depth = common;
        t[mid].l = t[old].l;
        t[mid].r = t[old].l + cut;
        t[mid].ch.push_back(old);
        t[p].ch.back() = mid;
        t[old].par = mid;
        t[old].l += cut;
        path.push_back(mid);
        p = mid;
      }

      int len = n - sa[i];
      if (len == common) {
        // この suffix は既存の頂点で終わる。
        t[p].suffix_pos = sa[i];
        suffix_node[sa[i]] = p;
      } else {
        int v = (int)t.size();
        t.emplace_back();
        t[v].par = p;
        t[v].depth = len;
        t[v].l = sa[i] + common;
        t[v].r = n;
        t[v].suffix_pos = sa[i];
        t[p].ch.push_back(v);
        suffix_node[sa[i]] = v;
        path.push_back(v);
      }
    }

    // 各部分木の SA 区間を再帰なしで求める。
    for (int v : postorder()) {
      int lo = n, hi = 0;
      if (t[v].suffix_pos != -1) {
        lo = rank[t[v].suffix_pos];
        hi = lo + 1;
      }
      for (int u : t[v].ch) {
        lo = std::min(lo, t[u].sa_l);
        hi = std::max(hi, t[u].sa_r);
      }
      t[v].sa_l = (lo == n ? 0 : lo);
      t[v].sa_r = hi;
    }
  }

  int size() const { return (int)t.size(); }
  int edge_len(int v) const { return t[v].r - t[v].l; }
  int count(int v) const { return t[v].sa_r - t[v].sa_l; }

  // {到達した頂点, その頂点への辺で一致した文字数} を返す。
  // 部分文字列として存在しなければ {-1, -1}。
  std::pair<int, int> locate(std::string_view pat) const {
    int v = root, i = 0;
    while (i < (int)pat.size()) {
      const auto& cs = t[v].ch;
      unsigned char c = (unsigned char)pat[i];
      auto it = std::lower_bound(cs.begin(), cs.end(), c,
          [&](int u, unsigned char x) {
            return (unsigned char)s[t[u].l] < x;
          });
      if (it == cs.end() || (unsigned char)s[t[*it].l] != c)
        return {-1, -1};

      int u = *it, j = 0;
      while (i < (int)pat.size() && t[u].l + j < t[u].r &&
             s[t[u].l + j] == pat[i]) {
        ++i;
        ++j;
      }
      if (i == (int)pat.size()) return {u, j};
      if (t[u].l + j < t[u].r) return {-1, -1};
      v = u;
    }
    return {root, 0};
  }

  // pat の出現回数（重なりを含む）。空文字列は |s|+1 回。
  int count(std::string_view pat) const {
    if (pat.empty()) return (int)s.size() + 1;
    auto [v, offset] = locate(pat);
    return v == -1 ? 0 : count(v);
  }

  // 子が親より先に現れる順序（木 DP 用）。
  std::vector<int> postorder() const {
    std::vector<int> order;
    order.reserve(t.size());
    order.push_back(root);
    for (int i = 0; i < (int)order.size(); ++i)
      for (int u : t[order[i]].ch) order.push_back(u);
    std::reverse(order.begin(), order.end());
    return order;
  }
};

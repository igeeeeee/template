template <typename T>
struct SlopeTrick {
  const T INF = std::numeric_limits<T>::max() / 3;

  T min_f;
  std::priority_queue<T, std::vector<T>, std::less<>> L;
  std::priority_queue<T, std::vector<T>, std::greater<>> R;
  T add_l, add_r;

 private:
  void push_R(const T& a) { R.push(a - add_r); }

  T top_R() const {
    if (R.empty())
      return INF;
    else
      return R.top() + add_r;
  }

  T pop_R() {
    T val = top_R();
    if (not R.empty()) R.pop();
    return val;
  }

  void push_L(const T& a) { L.push(a - add_l); }

  T top_L() const {
    if (L.empty())
      return -INF;
    else
      return L.top() + add_l;
  }

  T pop_L() {
    T val = top_L();
    if (not L.empty()) L.pop();
    return val;
  }

  std::size_t size() const { return L.size() + R.size(); }

 public:
  SlopeTrick() : min_f(0), add_l(0), add_r(0) {}

  struct Query {
    T lx, rx, min_f;
  };

  // return {最小化区間の左端, 右端, 最小値}
  Query query() const { return {top_L(), top_R(), min_f}; }

  // f(x) += a
  void add_all(const T& a) { min_f += a; }
  
  // add \_
  // f(x) += max(a-x, 0)
  void add_a_minus_x(const T& a) {
    min_f += std::max(T(0), a - top_R());
    push_R(a);
    push_L(pop_R());
  }

  // add _/
  // f(x) += max(x-a, 0)
  void add_x_minus_a(const T& a) {
    min_f += std::max(T(0), top_L() - a);
    push_L(a);
    push_R(pop_L());
  }

  // add \/
  // f(x) += |x-a|
  void add_abs(const T& a) {
    add_a_minus_x(a);
    add_x_minus_a(a);
  }

  // \/ -> \_
  // f_new(x) = min f(y) (y <= x)
  void clear_right() {
    while (not R.empty()) R.pop();
  }

  // \/ -> _/
  // f_new(x) = min f(y) (y >= x)
  void clear_left() {
    while (not L.empty()) L.pop();
  }

  // \/ -> \_/
  // f_new(x) = min f(y) (x-b <= y <= x-a)
  void shift(const T& a, const T& b) {
    assert(a <= b);
    add_l += a;
    add_r += b;
  }

  // \/. -> .\/
  // f_new(x) = f(x-a)
  void shift(const T& a) { shift(a, a); }

  // f(x) を取得（L, R を破壊する）
  T get(const T& x) {
    T ret = min_f;
    while (not L.empty()) {
      ret += std::max(T(0), pop_L() - x);
    }
    while (not R.empty()) {
      ret += std::max(T(0), x - pop_R());
    }
    return ret;
  }

  // f(x) += st.f(x)（st を破壊する）
  void merge(SlopeTrick& st) {
    if (st.size() > size()) {
      std::swap(st.L, L);
      std::swap(st.R, R);
      std::swap(st.add_l, add_l);
      std::swap(st.add_r, add_r);
      std::swap(st.min_f, min_f);
    }
    while (not st.R.empty()) {
      add_x_minus_a(st.pop_R());
    }
    while (not st.L.empty()) {
      add_a_minus_x(st.pop_L());
    }
    min_f += st.min_f;
  }
};
/*
 * Slope Trick（区分線形凸関数の管理）
 * ref: https://maspypy.com/slope-trick-1-%E8%A7%A3%E8%AA%AC%E7%B7%A8
 * パクリ元 https://ei1333.github.io/library/structure/others/slope-trick.hpp.html
 * verify https://atcoder.jp/contests/abc127/submissions/80430205
 * 初期状態: f(x) = 0
 *
 * 【使い方】
 * SlopeTrick<ll> st;
 * st.add_abs(3);                 // f(x) += |x-3|
 * st.add_abs(7);                 // f(x) += |x-7|
 * auto [l, r, v] = st.query();   // l=3, r=7, v=4
 * // [l,r]: 最小値を取る x の区間、v: 最小値
 *
 * 【操作】
 * add_all(a)         : f(x) += a
 * add_a_minus_x(a)   : f(x) += max(a-x, 0)
 * add_x_minus_a(a)   : f(x) += max(x-a, 0)
 * add_abs(a)         : f(x) += |x-a|
 * clear_right()      : f_new(x) = min_{y<=x} f(y)
 * clear_left()       : f_new(x) = min_{y>=x} f(y)
 * shift(a,b)         : f_new(x) = min_{x-b<=y<=x-a} f(y)
 * shift(a)           : f_new(x) = f(x-a)
 * query()            : 最小値と最小化区間 [lx,rx] を取得
 * get(x)             : f(x) を取得（破壊的）
 * merge(st)          : f(x) += st.f(x)（stを破壊）
 *
 * 【計算量】N: 折れ点数、K: 削除する折れ点数
 * add_all, shift, query : O(1)
 * add_a_minus_x, add_x_minus_a, add_abs : O(log(N+1))
 * clear_left/right    : O(K log(N+1))
 * get                 : O(N log(N+1))
 * merge               : O(M log(N+M+1))
 *                       M = min(2つの折れ点数)
 *
 * 【注意】
 * ・T は基本的に long long を推奨
 * ・query() の lx, rx は非有界の場合 -INF, INF になる
 * ・get(x) は L, R を空にするため、その後は元の関数を使えない
 * ・merge(st) は st を破壊する。自分自身との merge は不可
 */

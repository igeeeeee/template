// verify https://atcoder.jp/contests/joi2016ho/submissions/80201425
template <class S, auto op, auto e>
struct DisjointSparseTable {
    int _n = 0, lg = 0;
    vector<S> a;
    vector<vector<S>> d;

    DisjointSparseTable() = default;
    explicit DisjointSparseTable(int n) : DisjointSparseTable(vector<S>(n, e())) {}
    explicit DisjointSparseTable(const vector<S>& v) { init(v); }

    void init(const vector<S>& v) {
        a = v;
        _n = (int)a.size();
        lg = 0;
        while ((1LL << lg) < _n) ++lg;
        d.assign(lg, vector<S>(_n, e()));

        for (int k = 0; k < lg; ++k) {
            int half = 1 << k;
            int width = half << 1;
            for (int l = 0; l < _n; l += width) {
                int m = min(l + half, _n);
                int r = min(l + width, _n);

                // 前半（suffix）
                d[k][m - 1] = a[m - 1];
                for (int i = m - 2; i >= l; --i) {
                    d[k][i] = op(a[i], d[k][i + 1]);
                }

                // 後半（prefix）
                if (m < r) {
                    d[k][m] = a[m];
                    for (int i = m + 1; i < r; ++i) {
                        d[k][i] = op(d[k][i - 1], a[i]);
                    }
                }
            }
        }
    }

    S get(int p) const {
        assert(0 <= p && p < _n);
        return a[p];
    }

    S prod(int l, int r) const {
        assert(0 <= l && l <= r && r <= _n);
        if (l == r) return e();
        if (l + 1 == r) return a[l];
        int k = 31 - __builtin_clz((unsigned)(l ^ (r - 1)));
        return op(d[k][l], d[k][r - 1]);
    }

    S all_prod() const {
        return prod(0, _n);
    }

    template <class F>
    int max_right(int l, F f) const {
        assert(0 <= l && l <= _n);
        assert(f(e()));
        int ok = l, ng = _n + 1;
        while (ng - ok > 1) {
            int mid = ok + (ng - ok) / 2;
            if (f(prod(l, mid))) ok = mid;
            else ng = mid;
        }
        return ok;
    }

    template <class F>
    int min_left(int r, F f) const {
        assert(0 <= r && r <= _n);
        assert(f(e()));
        int ng = -1, ok = r;
        while (ok - ng > 1) {
            int mid = ng + (ok - ng) / 2;
            if (f(prod(mid, r))) ok = mid;
            else ng = mid;
        }
        return ok;
    }
};

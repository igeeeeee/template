// ref https://info.atcoder.jp/entry/algorithm_lectures/disjoint_sparse_table
//verify https://judge.yosupo.jp/submission/397469
template <class T>
struct DisjointSparseTable{
  vector<T> a;
  vector<vector<T>> table;
  ll siz = 0;
  ll h = 0;//高さ
  T e; //単位元 初期化に使われるやつ
  function<T(T,T)> op;
  DisjointSparseTable(){};
  DisjointSparseTable(vector<T> x,T _e,function<T(T,T)> _op):e(_e),op(_op),a(x){
    siz = 1;
    h = 0;
    while(siz < sz(x)){
      siz <<= 1;
      ++h;
    }
    while(sz(x) < siz)x.push_back(e);
    table.resize(h);
    rep(k,h){
      table[k].resize(siz);
      rep(i,1<<(h-k-1)){
        ll l = (1<<(k+1))*i;
        ll r = (1<<(k+1))*(i+1);
        ll m = l+(1<<k);
        {//前
          T val = e;
          rrep(j,m-1,l){
            val = op(x[j],val);
            table[k][j] = val;
          }
        }
        //後ろ
        {
          T val = e;
          reps(j,m,r){
            val = op(val,x[j]);
            table[k][j] = val;
          }
        }
      }
    }
    
  }

  T query(ll l, ll r){
    assert(l < r && 0 <= l && r <= siz);
    if(r == l+1){
      return a[l];
    }
    ll tp = topbit(l^(r-1));
    return op(table[tp][l],table[tp][r-1]);
  }
  
};

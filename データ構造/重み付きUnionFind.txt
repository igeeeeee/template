template<class T>
struct WeightUnionFind {
  ll _n;
  vector<ll> par;
  vector<ll> siz;
  vector<T> diff_weight;

  WeightUnionFind(ll n = 1)
      : _n(n), par(n, -1), siz(n, 1), diff_weight(n, T{}) {}

  ll leader(ll a) {
    assert(0 <= a && a < _n);
    if (par[a] == -1) return a;
    ll r = leader(par[a]);
    diff_weight[a] += diff_weight[par[a]]; // XORなら ^=
    return par[a] = r;
  }

  // 根からaまでの重み差
  T weight(ll a) {
    leader(a);
    return diff_weight[a];
  }

  ll size(ll a) {
    return siz[leader(a)];
  }

  // weight(b) - weight(a) = w となるようにmergeする
  // 矛盾がある場合false、そうでなければtrue
  bool merge(ll a, ll b, T w) {
    assert(0 <= a && a < _n);
    assert(0 <= b && b < _n);

    ll u = leader(a);
    ll v = leader(b);

    // すでに連結済みなら矛盾判定
    if (u == v) {
      // XORなら (diff_weight[b] ^ diff_weight[a]) == w
      return diff_weight[b] - diff_weight[a] == w; 
    }

    w += diff_weight[a]; // XORなら ^=
    w -= diff_weight[b]; // XORなら ^=

    if (siz[u] < siz[v]) {
      swap(u, v);
      w = -w; // XORならこの行を削除
    }

    par[v] = u;
    siz[u] += siz[v];
    diff_weight[v] = w;
    return true;
  }

  // weight(b) - weight(a)
  // a,bが同じ連結成分に属する必要がある
  T diff(ll a, ll b) {
    assert(same(a, b));
    return weight(b) - weight(a); // XORなら ^
  }

  bool same(ll a, ll b) {
    return leader(a) == leader(b);
  }

  vector<vector<ll>> groups() {
    map<ll, vector<ll>> mp;
    for (ll i = 0; i < _n; i++) {
      mp[leader(i)].push_back(i);
    }
    vector<vector<ll>> ans;
    for (auto &p : mp) {
      ans.push_back(p.second);
    }
    return ans;
  }
};

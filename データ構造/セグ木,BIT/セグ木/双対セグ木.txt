//双対セグ木
//fm(f,g)はgにfを作用させる（ACL準拠）
// vefiry https://onlinejudge.u-aizu.ac.jp/solutions/problem/DSL_2_D/review/11750609/T21140031/C++23
template<class M>
struct DualSegmentTree{
  using FM = function<M(M,M)>;
  ll n,siz,height;
  FM fm;
  const M em;//Mの単位元
  vector<M> lazy;

  DualSegmentTree(ll _n,FM _fm,M _em):n(_n),fm(_fm),em(_em){
    siz = 1;
    height = 0;
    while(siz < n){
      siz <<= 1;
      ++height;
    }
    lazy.resize(2*siz,em);
  }

  DualSegmentTree(vector<M> v,FM _fm,M _em):
    DualSegmentTree((ll)v.size(),_fm,_em){
    for(ll i=0;i<n;i++){
      lazy[siz+i] = v[i];
    }
  }

  void all_apply(ll p,M f){
    lazy[p] = fm(f,lazy[p]);
  }

  void push(ll p){
    all_apply(2*p,lazy[p]);
    all_apply(2*p+1,lazy[p]);
    lazy[p] = em;
  }

  void set(ll p,M x){
    assert(0 <= p && p < n);
    p += siz;
    for(ll i=height;i>=1;i--){
      push(p >> i);
    }
    lazy[p] = x;
  }

  M get(ll p){
    assert(0 <= p && p < n);
    p += siz;
    for(ll i=height;i>=1;i--){
      push(p >> i);
    }
    return lazy[p];
  }

  void apply(ll p,M f){
    assert(0 <= p && p < n);
    p += siz;
    for(ll i=height;i>=1;i--){
      push(p >> i);
    }
    all_apply(p,f);
  }

  void apply(ll l,ll r,M f){
    assert(0 <= l && l <= r && r <= n);
    if(l == r)return;
    l += siz;
    r += siz;

    for(ll i=height;i>=1;i--){
      if(((l >> i) << i) != l)push(l >> i);
      if(((r >> i) << i) != r)push((r-1) >> i);
    }

    while(l < r){
      if(l & 1)all_apply(l++,f);
      if(r & 1)all_apply(--r,f);
      l >>= 1;
      r >>= 1;
    }
  }
};

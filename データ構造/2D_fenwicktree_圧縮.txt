// 座標圧縮を用いた疎な2次元Fenwick Tree。
// opは結合律と交換律を満たし、e()は単位元、inv()は逆元。
// 点の座標は構築時に確定させる。値は構築後もオンライン更新可能。
// 同じ座標が複数ある場合、構築時にopで統合する。
// 計算量（Nは登録された点の数）：
//   構築 O(N log N)、set/apply/prod O(log^2 N)、
//   get O(log N)、count O(log^2 N)、all_prod O(1)、
//   空間 O(N log N)。
// 矩形の指定は半開区間 [lx,rx) × [ly,ry)。
template<class S, S (*op)(S,S), S (*e)(), S (*inv)(S),
         class Coord = long long>
struct fenwicktree2d {
  using point_type = pair<Coord,Coord>;

private:
  int n = 0;
  vector<Coord> xs;
  vector<point_type> points; // (x,y)順、重複なし
  vector<S> values;          // 各点の現在値
  vector<vector<Coord>> ys;  // 各XノードのY座標（重複可）
  vector<vector<S>> bit;     // Y方向のFenwick Tree
  S total = e();

  int x_index(Coord x) const {
    return int(lower_bound(xs.begin(),xs.end(),x)-xs.begin());
  }

  int point_index(Coord x, Coord y) const {
    auto it = lower_bound(points.begin(),points.end(),
                          point_type{x,y});
    assert(it != points.end() && *it == point_type(x,y));
    return int(it-points.begin());
  }

  // X方向は先頭r個、Y方向はy未満の点を集計。
  // inclusive=trueならY方向はy以下。
  S prefix_index(int r, Coord y, bool inclusive = false) const {
    S res = e();
    for(int k = r;k > 0;k -= k&-k) {
      int j = inclusive
        ? int(upper_bound(ys[k].begin(),ys[k].end(),y)
              -ys[k].begin())
        : int(lower_bound(ys[k].begin(),ys[k].end(),y)
              -ys[k].begin());

      for(;j > 0;j -= j&-j)
        res = op(res,bit[k][j]);
    }
    return res;
  }

  int count_prefix_index(int r, Coord y,
                         bool inclusive = false) const {
    int res = 0;
    for(int k = r;k > 0;k -= k&-k) {
      res += inclusive
        ? int(upper_bound(ys[k].begin(),ys[k].end(),y)
              -ys[k].begin())
        : int(lower_bound(ys[k].begin(),ys[k].end(),y)
              -ys[k].begin());
    }
    return res;
  }

  S rect_prod(int l, int r, Coord ly, Coord ry,
              bool ry_inclusive) const {
    S a = prefix_index(r,ry,ry_inclusive);
    S b = prefix_index(l,ry,ry_inclusive);
    S c = prefix_index(r,ly);
    S d = prefix_index(l,ly);
    return op(op(a,inv(b)),op(inv(c),d));
  }

  int rect_count(int l, int r, Coord ly, Coord ry,
                 bool ry_inclusive) const {
    return count_prefix_index(r,ry,ry_inclusive)
         - count_prefix_index(l,ry,ry_inclusive)
         - count_prefix_index(r,ly)
         + count_prefix_index(l,ly);
  }

  void apply_index(int id, S delta) {
    values[id] = op(values[id],delta);
    total = op(total,delta);

    int m = int(xs.size());
    int k = x_index(points[id].first)+1;
    Coord y = points[id].second;

    for(;k <= m;k += k&-k) {
      int j = int(lower_bound(ys[k].begin(),ys[k].end(),y)
                  -ys[k].begin())+1;

      for(;j < int(bit[k].size());j += j&-j)
        bit[k][j] = op(bit[k][j],delta);
    }
  }

  void build(const vector<point_type>& pts,
             const vector<S>* initial) {
    assert(!initial || pts.size() == initial->size());

    n = 0;
    xs.clear();
    points.clear();
    values.clear();
    total = e();

    // 座標のソート・重複統合。
    vector<int> order(pts.size());
    iota(order.begin(),order.end(),0);
    sort(order.begin(),order.end(),[&](int a,int b) {
      return pts[a] < pts[b];
    });

    for(int id : order) {
      S val = initial ? (*initial)[id] : e();
      if(!points.empty() && points.back() == pts[id]) {
        values.back() = op(values.back(),val);
      } else {
        points.push_back(pts[id]);
        values.push_back(val);
      }
    }

    n = int(points.size());

    for(auto [x,y] : points) xs.push_back(x);
    xs.erase(unique(xs.begin(),xs.end()),xs.end());

    int m = int(xs.size());
    ys.assign(m+1,{});
    bit.assign(m+1,{});
    for(int k = 1;k <= m;++k) bit[k].push_back(e());

    // Y座標の昇順に各X方向BITノードへ配布する。
    // 各ノードのYリストも自動的に昇順になる。
    order.resize(n);
    iota(order.begin(),order.end(),0);
    sort(order.begin(),order.end(),[&](int a,int b) {
      if(points[a].second != points[b].second)
        return points[a].second < points[b].second;
      return points[a].first < points[b].first;
    });

    for(int id : order) {
      auto [x,y] = points[id];
      total = op(total,values[id]);

      for(int k = x_index(x)+1;k <= m;k += k&-k) {
        ys[k].push_back(y);
        bit[k].push_back(values[id]);
      }
    }

    // 各ノードのY方向Fenwick Treeを線形構築。
    for(int k = 1;k <= m;++k) {
      int len = int(ys[k].size());
      for(int j = 1;j <= len;++j) {
        int p = j+(j&-j);
        if(p <= len)
          bit[k][p] = op(bit[k][p],bit[k][j]);
      }
    }
  }

public:
  fenwicktree2d() { build({},nullptr); }

  explicit fenwicktree2d(const vector<point_type>& pts) {
    build(pts,nullptr);
  }

  fenwicktree2d(const vector<point_type>& pts,
                const vector<S>& initial) {
    build(pts,&initial);
  }

  int size() const { return n; }

  S get(Coord x, Coord y) const {
    return values[point_index(x,y)];
  }

  void apply(Coord x, Coord y, S delta) {
    apply_index(point_index(x,y),delta);
  }

  void add(Coord x, Coord y, S delta) {
    apply(x,y,delta);
  }

  void set(Coord x, Coord y, S value) {
    int id = point_index(x,y);
    apply_index(id,op(value,inv(values[id])));
  }

  // (-inf,rx) × (-inf,ry)
  S prefix_prod(Coord rx, Coord ry) const {
    return prefix_index(x_index(rx),ry);
  }

  // [lx,rx) × [ly,ry)
  S prod(Coord lx, Coord rx, Coord ly, Coord ry) const {
    if(!(lx < rx) || !(ly < ry)) return e();
    return rect_prod(x_index(lx),x_index(rx),ly,ry,false);
  }

  // [lx,rx] × [ly,ry]
  S prod_inclusive(Coord lx, Coord rx,
                   Coord ly, Coord ry) const {
    if(rx < lx || ry < ly) return e();
    int r = int(upper_bound(xs.begin(),xs.end(),rx)
                -xs.begin());
    return rect_prod(x_index(lx),r,ly,ry,true);
  }

  // 登録された座標の個数（現在値が0の点も数える）。
  int count(Coord lx, Coord rx, Coord ly, Coord ry) const {
    if(!(lx < rx) || !(ly < ry)) return 0;
    return rect_count(x_index(lx),x_index(rx),ly,ry,false);
  }

  S all_prod() const { return total; }
};
